def apply(path: Path) -> dict:
    doc = json.loads(path.read_text(encoding="utf-8"))
    rels = doc["relations"]
    missing = set(V8) ^ set(rels)
    if missing:
        raise SystemExit(f"v8 and relations.json disagree on names: {sorted(missing)}")
    n_cond = n_stat = n_sec = 0
    for name, (cond, status) in V8.items():
        r = rels[name]
        if cond is not None:
            if r.get("condition"):
                raise SystemExit(f"{name}: v8 would overwrite an existing condition")
            r["condition"] = cond
            n_cond += 1
        old = r.get("status")
        if old and old != status:
            r["status_2026_09"] = old
        r["status"] = status
        n_stat += 1
        want = r["part"].split(" — ")[0]                      # e.g. "Part 2"
        sec = r["section"]
        if not sec.startswith(want.replace("Part ", "")) and not sec.startswith(want):
            r["section_as_parsed"] = r["section"]
            r["section"] = r["part"]
            n_sec += 1
    for name, extra in ALSO.items():
        rels[name]["also"] = extra
    doc["meta"]["v8"] = META_NOTE
    path.write_text(json.dumps(doc, ensure_ascii=False, indent=1), encoding="utf-8", newline="\n")
    return {"conditions_filled": n_cond, "statuses_written": n_stat, "sections_restored": n_sec,
            "rows_restored": len(ALSO)}
